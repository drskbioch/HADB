"""Fig 3a: Block B rows by name class (treemap)."""
import pandas as pd, squarify
from figures import style as S, figdata as FD, layout as L
FIG, LET = 3, 'a'
META = dict(chart='treemap', data=['release/name_class_table.csv'], filter='1,065 Block B rows by name class')
COL = {'species_resolved': S.MAIN, 'rank_genus_by_design': '#D9A441', 'non_taxon': '#8C8C8C', 'unresolved': S.TIER['T2']}
LAB = {'species_resolved': 'resolved to a species', 'rank_genus_by_design': 'genus only (by design)', 'non_taxon': 'non-taxon', 'unresolved': 'unresolved'}


def render():
    w, h = L.panel_size(FIG, LET); bg = S.DEFAULT_BG[FIG]; fig = S.new_fig(w, h, bg); ax = fig.add_axes([0.2 / w, 0.25 / h, (w - 0.4) / w, (h - 0.95) / h]); ax.axis('off')
    T = pd.read_csv(FD.REL / 'name_class_table.csv'); T = T[T['class'].isin(COL)].sort_values('block_b_rows', ascending=False)
    rects = squarify.squarify(squarify.normalize_sizes(T.block_b_rows.tolist(), 100, 100), 0, 0, 100, 100)
    for r, (_, row) in zip(rects, T.iterrows()):
        ax.add_patch(__import__('matplotlib').patches.Rectangle((r['x'], r['y']), r['dx'], r['dy'], facecolor=COL[row['class']], edgecolor=S.BACKGROUNDS[bg], lw=1.5))
        big = row.block_b_rows > 200; ax.text(r['x'] + r['dx'] / 2, r['y'] + r['dy'] / 2, '%s\n%d (%.1f%%)' % (LAB[row['class']], row.block_b_rows, row.pct_of_block_b_rows) if big else '%d' % row.block_b_rows, ha='center', va='center',
                                              fontsize=S.TICK - 0.5 if big else S.TICK - 2, color='white' if row['class'] != 'rank_genus_by_design' else '#1A1A1A', fontweight='bold')
    ax.set_xlim(0, 100); ax.set_ylim(0, 100)
    fig.text(0.25 / w, 0.12 / h, 'genus only = 72, non-taxon = 15, unresolved = 48 rows', fontsize=S.TICK - 2, color=S.muted(bg)); S.title(fig, 'Names of the Block B rows'); return fig, int(T.block_b_rows.sum())

"""Fig 3c: spelling similarity of the fuzzy / spelling matches accepted by decision (histogram + rug)."""
import numpy as np, pandas as pd
from figures import style as S, figdata as FD, layout as L
FIG, LET = 3, 'c'
META = dict(chart='histogram + rug', data=['derived/name_review.csv', 'derived/name_decisions_step6.csv', 'derived/pending_positives.csv', 'derived/unresolved_names_review_v3.csv'],
            filter='accepted non-exact name matches with a spelling similarity (abbreviations excluded: their similarity is an expansion, not a spelling score); one value per input name')


def render():
    w, h = L.panel_size(FIG, LET); bg = S.DEFAULT_BG[FIG]; fig = S.new_fig(w, h, bg); ax = S.axes_in(fig, left=0.55, bottom=0.55, right=0.15, top=0.5); v = {}
    a = pd.read_csv(FD.DER / 'name_review.csv'); a = a[a.decision == 'accept']; v.update({r.input: r.similarity for r in a.itertuples()})
    b = pd.read_csv(FD.DER / 'name_decisions_step6.csv'); b = b[(b.applied == 'Y') & b['sim'].notna()]; v.update({r.name_as_published: r.sim for r in b.itertuples() if r.name_as_published not in v})
    c = pd.read_csv(FD.DER / 'pending_positives.csv'); c = c[c.decision == 'accept']; v.update({r.gei_name: r.nearest1_similarity for r in c.itertuples() if r.gei_name not in v})
    d = pd.read_csv(FD.DER / 'unresolved_names_review_v3.csv', keep_default_na=False); d = d[(d.decision == 'accept') & (d.match_class == 'spelling_manual')]; v.update({r.input_name: float(r.similarity) for r in d.itertuples() if r.input_name not in v})
    x = np.array(list(v.values())) / 100.0; bins = np.arange(0.86, 1.0001, 0.01); ax.hist(x, bins=bins, color=S.MAIN, lw=0.4, edgecolor=S.BACKGROUNDS[bg]); ax.axvline(0.95, color=S.TIER['T2'], lw=S.LW_LINE, ls='--')
    ax.plot(x, np.full_like(x, -0.35), '|', color=S.fg(bg), ms=5, mew=0.5, clip_on=False); ax.set_xlim(0.86, 1.0); ax.set_ylim(0, None); ax.set_xlabel('Spelling similarity (0 to 1)'); ax.set_ylabel('Names (n)')
    ax.text(0.951, ax.get_ylim()[1] * 0.92, '0.95', color=S.TIER['T2'], fontsize=S.TICK - 1); fig.text(0.55 / w, 0.12 / h, '%d of %d below 0.95' % ((x < 0.95).sum(), len(x)), fontsize=S.TICK - 2, color=S.muted(bg))
    S.title(fig, 'Similarity of accepted matches'); return fig, len(x)

"""Fig 3d: reconciliation from Block B rows to evidence rows (waterfall)."""
import numpy as np, pandas as pd
from matplotlib.patches import Rectangle
from figures import style as S, figdata as FD, layout as L
FIG, LET = 3, 'd'
META = dict(chart='waterfall', data=['release/block_b_row_accounting.csv', 'derived/master_species_v2.csv', 'release/evidence.csv', 'release/species_element.csv'], filter='Block B only; units change between steps (rows, master rows, evidence rows, species x element pairs)')


def render():
    w, h = L.panel_size(FIG, LET); bg = S.DEFAULT_BG[FIG]; fig = S.new_fig(w, h, bg); ax = S.axes_in(fig, left=0.65, bottom=1.0, right=0.2, top=0.5)
    A = pd.read_csv(FD.REL / 'block_b_row_accounting.csv'); M = pd.read_csv(FD.DER / 'master_species_v2.csv', usecols=['evidence_type'], low_memory=False); E = FD.evidence_all(); Eb = E[E.evidence_type == 'block_b_label']
    n0, n1 = len(A), int((A.bucket == 'master').sum()); mrows = int((M.evidence_type == 'block_b_label').sum()); ev = len(Eb); prs = Eb.drop_duplicates(['species', 'element']).shape[0]
    steps = [('Block B rows', n0, 'total'), ('not in master\n(genus, non-taxon,\nunresolved, blank)', n1 - n0, 'delta'), ('rows in master', n1, 'total'), ('master rows\n(name x element)', mrows - n1, 'delta'), ('master rows', mrows, 'total'),
             ('one row per\ncitation', ev - mrows, 'delta'), ('evidence rows', ev, 'total'), ('merged to\nspecies x element', prs - ev, 'delta'), ('species x element\npairs', prs, 'total')]
    level = 0
    for i, (lab, v, kind) in enumerate(steps):
        if kind == 'total': ax.bar(i, v, color=S.MAIN, width=0.7, lw=0); level = v; ax.text(i, v + 60, format(v, ','), ha='center', fontsize=S.TICK - 1.5, color=S.fg(bg))
        else:
            lo, hi = (level, level + v) if v >= 0 else (level + v, level); ax.bar(i, hi - lo, bottom=lo, color=S.ACCENT if v >= 0 else S.TIER['T2'], width=0.7, lw=0); ax.text(i, hi + 60, ('+' if v >= 0 else '') + format(v, ','), ha='center', fontsize=S.TICK - 1.5, color=S.fg(bg)); level += v
    ax.set_xticks(range(len(steps))); ax.set_xticklabels([s[0] for s in steps], fontsize=S.TICK - 2.5, rotation=0); ax.set_ylabel('Count (n)'); ax.set_ylim(0, max(mrows, ev) * 1.12); S.title(fig, 'From Block B rows to evidence rows'); return fig, n0

"""Fig 3e: Gei XRF species by threshold class vs headline T1 species."""
import numpy as np, pandas as pd
from figures import style as S, figdata as FD, layout as L
FIG, LET = 3, 'e'
META = dict(chart='grouped bars', data=['release/evidence.csv', 'release/species_element.csv', 'release/counts.csv'], filter='species per element: Gei XRF evidence (headline view) with best threshold class high-range or standard; all headline T1 species')
RANK = {'high-range': 0, 'standard': 1, 'label-only': 2}


def render():
    w, h = L.panel_size(FIG, LET); bg = S.DEFAULT_BG[FIG]; fig = S.new_fig(w, h, bg); ax = S.axes_in(fig, left=0.55, bottom=0.75, right=0.12, top=0.5)
    E = FD.evidence(); G = E[(E.evidence_type == 'gei_xrf_taxon') & E.element.isin(FD.CORE)].assign(r=lambda d: d.threshold_class.map(RANK)); P = G.groupby(['sp', 'element']).r.min().reset_index(); C = FD.counts_headline(); x = np.arange(4); wd = 0.26
    hi = [int(((P.element == e) & (P.r == 0)).sum()) for e in FD.CORE]; st = [int(((P.element == e) & (P.r == 1)).sum()) for e in FD.CORE]; t1 = [FD.n_of(C, e, 'T1') for e in FD.CORE]
    for k, (vals, lab, al) in enumerate(((hi, 'Gei XRF, high-range', 1.0), (st, 'Gei XRF, standard only', 0.55), (t1, 'headline T1 (all sources)', 0.25))):
        for i, e in enumerate(FD.CORE): ax.bar(i + (k - 1) * wd, vals[i], width=wd * 0.92, color=S.elem_color(e), alpha=al, lw=0, label=lab if i == 0 else None)
        for i, v in enumerate(vals): ax.text(i + (k - 1) * wd, v + 3, str(v), ha='center', fontsize=S.TICK - 3, color=S.fg(bg))
    ax.set_xticks(x); ax.set_xticklabels(FD.CORE); ax.set_ylabel('Species (n)'); ax.legend(loc='upper right', handlelength=1, labelcolor=S.fg(bg)); S.title(fig, 'Thresholds in the Gei screen'); return fig, int(len(P))
#### fig3f.py
"""Fig 3f: what the headline view removes, by reason (dot plot)."""
import numpy as np, pandas as pd
from figures import style as S, figdata as FD, layout as L
FIG, LET = 3, 'f'
META = dict(chart='dot plot', data=['release/evidence.csv', 'release/species_element.csv'], filter='all evidence rows of rc3; pairs = species x element lost when every row of the pair is removed')


def render():
    w, h = L.panel_size(FIG, LET); bg = S.DEFAULT_BG[FIG]; fig = S.new_fig(w, h, bg); ax = S.axes_in(fig, left=1.05, bottom=0.7, right=0.2, top=0.5)
    E = FD.evidence_all(); flags = {'suspect source (CORE PDF)': E.suspect_source == 'Y', 'weakened (red, not confirmed)': E.weakened == 'Y', 'inferred citation (DOI replaced)': E.citation_inferred == 'Y', 'any reason': ~E.headline}
    g = E.assign(rm=~E.headline).groupby(['species', 'element']).rm.all(); lost = set(g[g].index); rows = []
    for k, m in flags.items():
        gg = E.assign(m=m).groupby(['species', 'element']).m.all(); rows.append((k, int(m.sum()), len(set(gg[gg].index) & lost) if k != 'any reason' else len(lost)))
    D = pd.DataFrame(rows, columns=['reason', 'rows', 'pairs']).iloc[::-1]; y = np.arange(len(D))
    ax.hlines(y, D.pairs.clip(lower=0.8), D.rows, color=S.GREY, lw=S.LW_LINE); ax.scatter(D.rows, y, s=24, color=S.MAIN, zorder=3, label='evidence rows'); ax.scatter(D.pairs.clip(lower=0.8), y, s=24, color=S.TIER['T2'], zorder=3, label='pairs lost')
    for yi, a, b in zip(y, D.rows, D.pairs): ax.text(max(a, b) * 1.3, yi, '%d / %d' % (a, b), va='center', fontsize=S.TICK - 2.5, color=S.fg(bg))
    ax.set_xscale('log'); ax.set_xlim(0.7, 2000); ax.set_yticks(y); ax.set_yticklabels(D.reason, fontsize=S.TICK - 2.5); ax.set_xlabel('Count (n, log axis)'); ax.legend(loc='lower right'); ax.grid(axis='x', color=S.grid_color(bg), lw=S.LW_GRID); ax.set_axisbelow(True)
    S.title(fig, 'Removed from the headline view'); return fig, len(E)
#### fig3g.py
"""Fig 3g: licence status of the evidence rows (donut)."""
import numpy as np
from figures import style as S, figdata as FD, layout as L
FIG, LET = 3, 'g'
META = dict(chart='donut', data=['release/evidence.csv'], filter='all evidence rows of rc3; licence recorded for the source of each row')
COL = {'CC BY': S.MAIN, 'CC BY-NC': '#D9A441', 'CC BY-NC-ND': S.TIER['T2'], 'not stated or unknown': '#B8B8B8'}


def lic(s):
    s = s.lower()
    return 'CC BY-NC-ND' if 'by-nc-nd' in s else 'CC BY-NC' if 'by-nc' in s else 'CC BY' if 'cc by' in s else 'not stated or unknown'


def render():
    w, h = L.panel_size(FIG, LET); bg = S.DEFAULT_BG[FIG]; fig = S.new_fig(w, h, bg); ax = fig.add_axes([0.3 / w, 1.0 / h, (w - 0.6) / w, (h - 1.6) / h])
    E = FD.evidence_all(); c = E.source_license.map(lic).value_counts().reindex(list(COL)).fillna(0); ax.pie(c.values, colors=[COL[k] for k in c.index], startangle=90, counterclock=False, wedgeprops=dict(width=0.38, edgecolor=S.BACKGROUNDS[bg], linewidth=1))
    ax.text(0, 0, '%s\nrows' % format(len(E), ','), ha='center', va='center', fontsize=S.TICK, color=S.fg(bg), fontweight='bold')
    for j, (k, v) in enumerate(c.items()):
        fig.add_artist(__import__('matplotlib').patches.Rectangle((0.45 / w, (0.78 - 0.17 * j) / h), 0.1 / w, 0.1 / h, transform=fig.transFigure, color=COL[k], lw=0)); fig.text(0.62 / w, (0.83 - 0.17 * j) / h, '%s  %s (%.0f%%)' % (k, format(int(v), ','), 100 * v / len(E)), fontsize=S.TICK - 2, color=S.fg(bg), va='center')
    S.title(fig, 'Licence of the evidence sources'); return fig, len(E)
